    // Use damping for numerical stability
    double alpha = 0.3; // Start with 0.3
    Vector delta = jbody.pseudoInverse().multiply(se3algebra.se3ToVec(bodyse3alg));
    thetaList = thetaList.add(delta.scalarMult(alpha));
    
    // Check if we're making progress
    se3group newTsb = forwardKinematics.FKinBody(M, Blist, thetaList, verbose);
    se3group newX = new se3group(newTsb.matrix.pseudoInverse().multiply(T.matrix));
    se3algebra newLog = se3ops.matrixLog6(newX, false);
    Vector6 newTwist = se3algebra.se3ToVec(newLog);
    double newError = newTwist.norm();
    double oldError = se3algebra.se3ToVec(bodyse3alg).norm();
    
    // If error increased, reduce step size
    if (newError > oldError) {
        alpha *= 0.5;
        thetaList = thetaList.add(delta.scalarMult(alpha));
        log += "Reduced step size to: " + alpha + "\n";
    }
    
    Wb = bodyse3alg.getOmega();
    Vb = bodyse3alg.getVelocity();
    iteration++;
    
    log += "iteration=" + iteration + ", error=" + newError + "\n";
    
} while ((Wb.norm() > eomg || Vb.norm() > ev) && iteration < maxIterations);
